def run(label, case, precision, backend="CPU", probe=False):
    directory = output / label
    command = [str(blas if backend == "BLAS" else cpu), str(models[precision]), str(conditions[case]),
               str(directory), "20", "1.5", "1.5", "1", "42", "1" if case == "cardigan" else "2", "16",
               "--backend", backend, "--matrix-type", "bf16" if precision == "bf16" else "q8_0",
               "--mmap", "off", "--load-threads", "2", "--precompute-modulations",
               "--modulation-cache-mib", "128", "--fused-gelu", "--memory-profile"]
    if precision == "bf16":
        command.append("--upcast-matrices")
    if precision == "mixed":
        command += ["--f32-matrices", str(policy)]
    if probe:
        command += ["--probe-forwards", "3", "--no-record"]
    row = report["runs"].setdefault(label, {"command": command, "completed": False})
    if row["command"] != command:
        raise RuntimeError("Resume command changed")
    save()
    if "measurement" not in row:
        if directory.exists():
            raise RuntimeError(f"Partial run {label} needs inspection; refusing automatic regeneration")
        row["measurement"] = measure_command(command, output / (label + ".log"), 7200,
                                              output / (label + "-timeline.jsonl"))
        save()
    if row["measurement"]["exit_code"]:
        raise RuntimeError(f"{label} process failed")
    row["manifest"] = json.loads((directory / "manifest.json").read_text())
    profile = [json.loads(line) for line in (directory / "memory-profile.jsonl").read_text().splitlines()]
    if not profile or profile[-1]["phase"] != "model_released":
        raise RuntimeError("Incomplete model lifetime profile")
    if any(item.get("conversion", {}).get("converted_tensors", 0) for item in profile):
        raise RuntimeError("Runtime-ready inference unexpectedly converted weights")
    row["manifest_sha256"] = sha256(directory / "manifest.json")
    row["memory_profile_sha256"] = sha256(directory / "memory-profile.jsonl")
    row["working_set_ceiling_bytes"] = (3 if precision == "bf16" else 2) * 1024**3
    if row["measurement"]["peak_working_set_bytes"] > row["working_set_ceiling_bytes"]:
        raise RuntimeError(f"{label} exceeded declared process working-set ceiling")
    row["completed"] = True
    save()
    print(label, "execution complete", row["manifest"]["sampling_seconds"],
          row["measurement"]["peak_working_set_bytes"], flush=True)
    return directory, row


def compare_recordings(reference, actual, exact):
    rm = json.loads((reference / "manifest.json").read_text())
    nm = json.loads((actual / "manifest.json").read_text())
    for key in ("steps", "cfg", "shift", "skip_cfg_last_n_steps", "seed", "category", "threads",
                "schedule", "matrix_type", "upcast_matrices", "matrix_types"):
        if key not in rm or rm[key] != nm[key]:
            raise RuntimeError(f"Native recording setting mismatch: {key}")
    result = {"passed": True, "exact_required": exact, "files": [],
              "reference_manifest_sha256": sha256(reference / "manifest.json")}
    for filename in ["initial.safetensors"] + [f"step-{i}.safetensors" for i in range(20)]:
        before, after = reference / filename, actual / filename
        reference_tensors, actual_tensors = load_file(str(before)), load_file(str(after))
        if set(reference_tensors) != set(actual_tensors):
            raise RuntimeError("Incomplete native recording")
        entry = {"file": filename, "reference_sha256": sha256(before), "actual_sha256": sha256(after), "tensors": {}}
        for key, expected in reference_tensors.items():
            value = actual_tensors[key]
            metric = metrics(expected, value)
            identical = canonical(expected).tobytes() == canonical(value).tobytes()
            metric["bitwise_equal"] = identical
            if filename == "initial.safetensors":
                metric["passed"] = metric["max_abs"] <= 1e-6
            if exact:
                metric["passed"] = metric["passed"] and identical
            result["passed"] = result["passed"] and metric["passed"]
            entry["tensors"][key] = metric
        result["files"].append(entry)
    return result


def cropped_image(directory):
    image = pixels(load_file(str(directory / "step-19.safetensors"))["image"])
    return image[48:816, :576]


def contact_sheet(label, expected, actual):
    delta = actual.astype(np.int16) - expected.astype(np.int16)
    result = {"changed_channels": int(np.count_nonzero(delta)), "max_abs": int(np.abs(delta).max()),
              "mae": float(np.abs(delta).mean()), "exact": bool(np.array_equal(expected, actual))}
    canvas = Image.new("RGB", (576 * 3, 808), "white")
    draw = ImageDraw.Draw(canvas)
    for i, (name, array) in enumerate((("Reference", expected), ("Optimized", actual),
                                        ("Absolute difference x32", np.clip(np.abs(delta) * 32, 0, 255).astype(np.uint8)))):
        canvas.paste(Image.fromarray(array), (576 * i, 40))
        draw.text((576 * i + 10, 12), name, fill="black")
    canvas.save(output / (label + "-comparison.png"))
    Image.fromarray(actual).save(output / (label + ".png"))
    return result
