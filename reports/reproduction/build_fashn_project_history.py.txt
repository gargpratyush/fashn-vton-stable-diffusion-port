def esc(value):
    return html.escape(str(value), quote=True)


def read(path):
    path = Path(path)
    data = path.read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    key = str(path)
    if key in records:
        assert records[key]["sha256"] == digest, "Source changed during compilation: " + key
    records[key] = {"sha256": digest, "bytes": len(data)}
    return data


def text(path):
    return read(path).decode("utf-8-sig")


def link(path, label=None):
    path = Path(path)
    assert path.exists(), path
    return f'<a href="{esc(path.as_uri())}">{esc(label or path.name)}</a>'


def paras(value):
    return "".join("<p>" + esc(p.strip()) + "</p>" for p in value.strip().split("\n\n") if p.strip())


def table(headers, rows):
    return '<div class="table-scroll"><table><thead><tr>' + "".join(
        "<th>" + esc(v) + "</th>" for v in headers) + "</tr></thead><tbody>" + "".join(
        "<tr>" + "".join("<td>" + esc(v) + "</td>" for v in row) + "</tr>" for row in rows) + "</tbody></table></div>"


def example(label, content, note="Explanatory sketch, not a complete standalone program."):
    return f'<div class="code-label">{esc(label)}</div><p class="small">{esc(note)}</p><pre><code>{esc(content.strip())}</code></pre>'


def source(relative, first, last, explanation):
    path = REPO / relative
    lines = text(path).splitlines()
    assert 1 <= first <= last <= len(lines), (path, first, last)
    body = "\n".join(lines[first - 1:last])
    identity = {"path": str(path), "first_line": first, "last_line": last,
                "excerpt_sha256": hashlib.sha256(body.encode()).hexdigest()}
    snippets.append(identity)
    numbered = "".join(f'<span class="code-line"><span class="line-no">{i}</span>{esc(line)}</span>'
                       for i, line in enumerate(lines[first - 1:last], first))
    return (f'<div class="source"><div class="code-label">Current source: {link(path, relative)}'
            f' &middot; L{first}&ndash;{last}</div>{paras(explanation)}'
            f'<pre><code>{numbered}</code></pre></div>')


def image(path, caption):
    path = Path(path)
    data = read(path)
    suffix = path.suffix.lower()
    mime = {".png": "image/png", ".jpg": "image/jpeg", ".jpeg": "image/jpeg"}[suffix]
    illustrations.append({"path": str(path), "sha256": records[str(path)]["sha256"], "caption": caption})
    uri = f"data:{mime};base64," + base64.b64encode(data).decode("ascii")
    return f'<figure><img loading="lazy" src="{uri}" alt="{esc(caption)}"><figcaption>{esc(caption)} Right-click the image to save its embedded copy. {link(path, "Local original")}</figcaption></figure>'


def chapter(number, title, why, how, result, lesson, code="", extra="", evidence=(), status="Delivered"):
    identifier = f"cp-{number:02d}"
    body = (f'<article id="{identifier}" class="chapter"><div class="eyebrow">Checkpoint {number:02d}'
            f' <span class="badge">{esc(status)}</span></div><h2>{esc(title)}</h2>'
            f'<h3>Why this step was needed</h3>{paras(why)}<h3>What changed and how</h3>{paras(how)}'
            f'{code}<h3>Observed results</h3>{paras(result)}{extra}'
            f'<div class="lesson"><strong>What we learned / boundary</strong>{paras(lesson)}</div>')
    if evidence:
        body += '<p class="evidence"><strong>Evidence:</strong> ' + " &middot; ".join(
            link(ROOT / p, p) for p in evidence) + "</p>"
    body += "</article>"
    chapters.append({"number": number, "id": identifier, "title": title, "status": status, "html": body})

def inline_markdown(value, base):
    tokens = []

    def stash(fragment):
        tokens.append(fragment)
        return f"\x00{len(tokens)-1}\x00"

    value = re.sub(r"`([^`]+)`", lambda m: stash("<code>" + esc(m.group(1)) + "</code>"), value)

    def md_link(match):
        label, target = match.groups()
        if target.startswith(("https://", "http://")):
            href = target
        elif target.startswith("#"):
            return label + " (" + target + ")"
        else:
            path = Path(target)
            if not path.is_absolute():
                path = base / target.replace("/", "\\")
            if not path.exists():
                return label + " [" + target + "; historical link]"
            href = path.resolve().as_uri()
        return stash(f'<a href="{esc(href)}">{esc(label)}</a>')

    value = re.sub(r"\[([^\]]+)\]\(([^)]+)\)", md_link, value)
    value = esc(value)
    value = re.sub(r"\*\*([^*]+)\*\*", r"<strong>\1</strong>", value)
    for i in reversed(range(len(tokens))):
        value = value.replace(f"\x00{i}\x00", tokens[i])
    return value


def render_markdown(value, base):
    """Render the report subset; preserve the exact original alongside it."""
    lines = value.splitlines()
    output = []
    i = 0
    while i < len(lines):
        line = lines[i]
        stripped = line.strip()
        if not stripped:
            i += 1
            continue
        if stripped.startswith("```"):
            block = []
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("```"):
                block.append(lines[i])
                i += 1
            output.append("<pre><code>" + esc("\n".join(block)) + "</code></pre>")
            i += 1
            continue
        heading = re.match(r"^(#{1,6})\s+(.+)", stripped)
        if heading:
            level = min(6, len(heading[1]) + 2)
            output.append(f"<h{level}>" + inline_markdown(heading[2], base) + f"</h{level}>")
            i += 1
            continue
        if i + 1 < len(lines) and "|" in line and re.fullmatch(r"[\s|:\-]+", lines[i + 1]) and "-" in lines[i + 1]:
            headers = [v.strip() for v in line.strip().strip("|").split("|")]
            rows = []
            i += 2
            while i < len(lines) and lines[i].strip().startswith("|"):
                rows.append([v.strip() for v in lines[i].strip().strip("|").split("|")])
                i += 1
            output.append('<div class="table-scroll"><table><thead><tr>' + "".join(
                "<th>" + inline_markdown(v, base) + "</th>" for v in headers) + "</tr></thead><tbody>" +
                "".join("<tr>" + "".join("<td>" + inline_markdown(v, base) + "</td>" for v in row) + "</tr>" for row in rows) +
                "</tbody></table></div>")
            continue
        if re.match(r"^([-*]|\d+[.])\s+", stripped):
            items = []
            while i < len(lines) and re.match(r"^\s*([-*]|\d+[.])\s+", lines[i]):
                item = re.sub(r"^\s*([-*]|\d+[.])\s+", "", lines[i])
                i += 1
                while i < len(lines) and lines[i].startswith(("  ", "\t")) and not re.match(r"^\s*([-*]|\d+[.])\s+", lines[i]):
                    item += " " + lines[i].strip()
                    i += 1
                items.append("<li>" + inline_markdown(item, base) + "</li>")
            output.append("<ul>" + "".join(items) + "</ul>")
            continue
        paragraph = [stripped]
        i += 1
        while i < len(lines) and lines[i].strip() and not re.match(r"^\s*(#{1,6}\s|```|[-*]\s|\d+[.]\s|\|)", lines[i]):
            paragraph.append(lines[i].strip())
            i += 1
        output.append("<p>" + inline_markdown(" ".join(paragraph), base) + "</p>")
    return "".join(output)
