find_package(OpenCV 4.12 REQUIRED COMPONENTS core imgproc imgcodecs)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C/C++ distribution directory")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include")
find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib")
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "SD_FASHN_PREPROCESS requires ONNX Runtime 1.22.1 C/C++ headers and library; set ONNXRUNTIME_ROOT")
endif()
add_library(fashn-preprocessing STATIC preprocess.cpp pipeline.cpp)
target_compile_features(fashn-preprocessing PUBLIC cxx_std_17)
target_include_directories(fashn-preprocessing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" ${OpenCV_INCLUDE_DIRS} "${ONNXRUNTIME_INCLUDE_DIR}"
    PRIVATE "${PROJECT_SOURCE_DIR}/thirdparty")
target_link_libraries(fashn-preprocessing PUBLIC ${OpenCV_LIBS} "${ONNXRUNTIME_LIBRARY}")
add_executable(sd-fashn-prepare main.cpp)
target_include_directories(sd-fashn-prepare PRIVATE "${PROJECT_SOURCE_DIR}/thirdparty")
target_link_libraries(sd-fashn-prepare PRIVATE fashn-preprocessing)
set(FASHN_RUNTIME_DLLS "")
if(WIN32)
    target_compile_definitions(fashn-preprocessing PRIVATE NOMINMAX)
    target_link_libraries(fashn-preprocessing PUBLIC bcrypt)
    list(APPEND FASHN_RUNTIME_DLLS "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll")
    install(FILES "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" DESTINATION bin COMPONENT fashn-preprocess)
    set(FASHN_OPENCV_RUNTIME_LIBS ${OpenCV_LIBS})
    list(REMOVE_DUPLICATES FASHN_OPENCV_RUNTIME_LIBS)
    foreach(opencv_lib IN LISTS FASHN_OPENCV_RUNTIME_LIBS)
        if(TARGET ${opencv_lib} AND OpenCV_SHARED)
            get_target_property(is_imported ${opencv_lib} IMPORTED)
            if(is_imported)
                set_property(TARGET ${opencv_lib} PROPERTY IMPORTED_GLOBAL TRUE)
            endif()
            list(APPEND FASHN_RUNTIME_DLLS "$<TARGET_FILE:${opencv_lib}>")
            install(FILES "$<TARGET_FILE:${opencv_lib}>" DESTINATION bin COMPONENT fashn-preprocess)
        endif()
    endforeach()
else()
    find_package(OpenSSL REQUIRED)
    target_link_libraries(fashn-preprocessing PUBLIC OpenSSL::Crypto)
endif()
set_property(TARGET fashn-preprocessing PROPERTY FASHN_RUNTIME_DLLS "${FASHN_RUNTIME_DLLS}")
function(fashn_copy_runtime target)
    get_target_property(runtime_dlls fashn-preprocessing FASHN_RUNTIME_DLLS)
    foreach(dll IN LISTS runtime_dlls)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
endfunction()
fashn_copy_runtime(sd-fashn-prepare)
install(TARGETS sd-fashn-prepare RUNTIME DESTINATION bin COMPONENT fashn-preprocess)
